// 粒子运动模拟：物理模型
class Particle {
  constructor(x, y) {
    this.x = x;
    this.y = y;
    this.vx = Math.random() * 4 - 2;
    this.vy = Math.random() * 4 - 2;
    this.life = 100;
  }
  update() {
    this.x += this.vx;
    this.y += this.vy;
    this.life--;
  }
  draw(ctx) {
    ctx.globalAlpha = this.life / 100;
    ctx.beginPath();
    ctx.arc(this.x, this.y, 3, 0, Math.PI * 2);
    ctx.fill();
  }
}

function nextGeneration(grid){
  return grid.map((row,y)=>
   row.map((cell,x)=>{
     let neighbors=countNeighbors(grid,x,y);
    if (cell && (neighbors === 2 || neighbors === 3)) return 1; // 保持活着
    if (!cell && neighbors === 3) return 1;                    // 繁衍新生
    return 0;                                                  // 其他情况全部死亡
   })
 );
}

//斐波那契数列
function countRabbits(months) {
  let young = 1; // 小兔子
  let adult = 0; // 大兔子

  for (let month = 1; month <= months; month++) {
    const baby = adult;       // 1. 每只大兔子生一只小兔子
    adult = adult + young;    // 2. 原来的小兔子长成了大兔子
    young = baby;             // 3. 刚出生的小兔子进入幼兔池
  }

  return young + adult;
}
